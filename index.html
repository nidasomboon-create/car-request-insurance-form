หัวตารางแถวที่ 1 ใน Google Sheets ของคุณควรตรงตามนี้:
Timestamp | Name | Brand | Model | Year | Phone | ContactOthers
/* Apps Script Code */ function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.fullName, data.carBrand, data.carModel, data.carYear, data.contactPhone, data.contactLine ]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput("เชื่อมต่อระบบสำเร็จ!"); }
